#   OMP_NUM_THREADS=1

#   OPENBLAS_NUM_THREADS=1

# Common flags:
#   --n-envs 48
#   --n-steps 1024
#   --batch-size 256
#   --n-epochs 4
#   --lr 3e-4
#   --gamma 0.995
#   --gae-lambda 0.95

#   --episode-time-limit 5000.0
#   --blue-budget 200000000
#   --red-budget 150000000
#   --device cpu
#   --save-every 200000
#
# INCREMENT pattern: --total-steps 200000 per call, then convergence check;
# advance to next stage when last 3 ep_rew_mean values span < 5.0 AND
# the latest is not a new global max. HARD_CAP advances regardless.
#
# Stage S1 (Layout C, scripted Red, cold start):

#     --red-mode-pool scripted --warmstart .../stages/S1_final.zip

#     --red-mode-pool scripted scripted rule --warmstart .../stages/S2_final.zip
